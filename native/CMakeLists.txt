cmake_minimum_required(VERSION 3.20)
project(BeatCheatProDSP LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
add_library(beatcheatpro_dsp SHARED src/detector.cpp)
target_include_directories(beatcheatpro_dsp PUBLIC include)
if(MSVC)
  target_compile_options(beatcheatpro_dsp PRIVATE /O2)
else()
  target_compile_options(beatcheatpro_dsp PRIVATE -O3)
endif()
